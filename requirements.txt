# EduMusic dependencies. Install with:
#
#     ./setup.sh
#
# or, inside an activated virtualenv:
#
#     pip install -r requirements.txt
#
# Everything here ships prebuilt wheels -- no compiler needed.

# Audio I/O (microphone input + metronome playback)
sounddevice>=0.4.6

# Array math for pitch conversion and click synthesis
numpy>=1.24

# Monophonic pitch detection (YIN). Replaced aubio, which has no macOS arm64
# wheels and whose last release (0.4.9, 2019) no longer compiles against
# current clang or ffmpeg. librosa is maintained and installs from wheels.
# It is a heavier dependency tree (numba, scipy, scikit-learn), but detection
# costs ~0.25 ms per frame against an 11.6 ms budget, so speed is not at issue.
librosa>=0.10
